cmake_minimum_required(VERSION 3.24)
project(RotaryMelterNative VERSION 3.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(ROTARY_DEPENDENCIES "${CMAKE_CURRENT_SOURCE_DIR}/../.tools/cpp_deps" CACHE PATH "Pinned numerical headers")
set(ROTARY_SOURCE_FINGERPRINT "development" CACHE STRING "SHA-256 of the compiled source set")
if(NOT EXISTS "${ROTARY_DEPENDENCIES}/nlohmann/json.hpp")
  message(FATAL_ERROR "Run the local virtual environment Python with -m tools.SetupCpp first")
endif()
find_package(Threads REQUIRED)
find_package(OpenMP QUIET)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(rotary_melter_native SHARED src/Configuration.cpp src/Thermodynamics.cpp src/Geometry.cpp src/Calculation.cpp src/Analysis.cpp src/ContactDynamics.cpp src/MotionApi.cpp src/Api.cpp)
if(WIN32)
  set_target_properties(rotary_melter_native PROPERTIES PREFIX "")
endif()
target_include_directories(rotary_melter_native PUBLIC include "${ROTARY_DEPENDENCIES}" "${ROTARY_DEPENDENCIES}/eigen")
target_link_libraries(rotary_melter_native PRIVATE Threads::Threads)
target_compile_definitions(rotary_melter_native PRIVATE ROTARY_NATIVE_BUILD EIGEN_DONT_PARALLELIZE "ROTARY_SOURCE_FINGERPRINT=\"${ROTARY_SOURCE_FINGERPRINT}\"")
if(OpenMP_CXX_FOUND)
  target_link_libraries(rotary_melter_native PRIVATE OpenMP::OpenMP_CXX)
endif()
if(MSVC)
  target_compile_options(rotary_melter_native PRIVATE /W4 /utf-8)
else()
  target_compile_options(rotary_melter_native PRIVATE -Wall -Wextra -Wpedantic)
  if(WIN32)
    target_link_options(rotary_melter_native PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
add_executable(rotary_melter_cpp src/main.cpp)
target_link_libraries(rotary_melter_cpp PRIVATE rotary_melter_native)
if(WIN32)
  target_link_libraries(rotary_melter_cpp PRIVATE shell32)
endif()
enable_testing()
add_test(NAME native_backend_info COMMAND rotary_melter_cpp --info)
